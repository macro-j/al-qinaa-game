CRITICAL: Configure the SpeechSynthesisUtterance to use Arabic!

JavaScript
const speak = (text) => {
  if ('speechSynthesis' in window) {
    const utterance = new SpeechSynthesisUtterance(text);
    utterance.lang = 'ar-SA'; // Strict Arabic
    utterance.rate = 0.8; // Slightly slow for a dramatic narrator effect
    utterance.pitch = 0.7; // Deeper voice
    window.speechSynthesis.speak(utterance);
  }
};
